# Compile the identical desktop shell once for the application and native
# regression harnesses. Each harness owns only its own main/test source.
set(EMCI_SHELL_SOURCES src/calculatortools.cpp src/desktopsurface.cpp src/mainwindow.cpp
 src/mainwindow_workspace.cpp src/graphpanel.cpp src/i18n.cpp src/calculatorpanel.cpp
 src/calculatorkeyboard.cpp src/calculatormenu.cpp)
add_library(emci_desktop_shell OBJECT ${EMCI_SHELL_SOURCES})
target_link_libraries(emci_desktop_shell PRIVATE emci_workspace emci_history emci_currency emci_registers emci_finance_ui Qt6::Widgets)
get_property(emci_shell_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(t IN LISTS emci_shell_targets)
 if(t STREQUAL "emci_desktop_shell")
  continue()
 endif()
 get_target_property(sources ${t} SOURCES)
 if("src/mainwindow.cpp" IN_LIST sources)
  foreach(common IN LISTS EMCI_SHELL_SOURCES)
   list(REMOVE_ITEM sources "${common}")
  endforeach()
  set_property(TARGET ${t} PROPERTY SOURCES "${sources};$<TARGET_OBJECTS:emci_desktop_shell>")
 endif()
endforeach()
